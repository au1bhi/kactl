\section{Derangements }
	Permutations of a set such that none of the elements appear in their original position.
	\[ D(n) = (n-1)(D(n-1)+D(n-2)) = n D(n-1)+(-1)^n = \left\lfloor\frac{n!}{e}\right\rceil \]

\section{Lucas' Theorem}
	Let $n,m$ be non-negative integers and $p$ a prime. Write $n=n_kp^k+...+n_1p+n_0$ and $m=m_kp^k+...+m_1p+m_0$. Then 
	\[ \binom{n}{m} \equiv \prod_{i=0}^k\binom{n_i}{m_i} \pmod{p} \]

\section{Stirling numbers of the second kind}
	Partitions of $n$ distinct elements into exactly $k$ non-empty unlabelled groups.
	$$S(n,k) = S(n-1,k-1) + k S(n-1,k)$$
	$$S(n,1) = S(n,n) = 1$$
	$$S(n,k) = \frac{1}{k!}\sum_{j=0}^k (-1)^{k-j}\binom{k}{j}j^n$$

\section{Bell numbers}
	Total number of partitions of $n$ distinct elements. $B(n) =$
	$1, 1, 2, 5, 15, 52, 203, 877, 4140, 21147, \dots$. For $p$ prime,
	\[ B(p^m+n)\equiv mB(n)+B(n+1) \pmod{p} \]

\section{Labeled unrooted trees (Cayley's Formula)}
	\# on $n$ vertices: $n^{n-2}$ \\
	\# on $k$ existing trees of size $n_i$: $n_1n_2\cdots n_k n^{k-2}$ \\
	\# with degrees $d_i$: $(n-2)! / ((d_1-1)! \cdots (d_n-1)!)$

\section{Catalan numbers}
	\[ C_n=\frac{1}{n+1}\binom{2n}{n}= \binom{2n}{n}-\binom{2n}{n+1} = \frac{(2n)!}{(n+1)!n!} \]
	\[ C_0=1,\ C_{n+1} = \frac{2(2n+1)}{n+2}C_n,\ C_{n+1}=\sum C_iC_{n-i} \]
	${C_n = 1, 1, 2, 5, 14, 42, 132, 429, 1430, 4862, 16796, 58786, \dots}$
	\begin{itemize}[noitemsep]
		\item sub-diagonal monotone paths in an $n\times n$ grid.
		\item strings with $n$ pairs of parenthesis, correctly nested.
		\item binary trees with with $n+1$ leaves (0 or 2 children).
		\item ordered trees with $n+1$ vertices.
		\item ways a convex polygon with $n+2$ sides can be cut into triangles.
		\item permutations of $[n]$ with no 3-term increasing subseq.
	\end{itemize}
